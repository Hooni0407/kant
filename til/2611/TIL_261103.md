### 딕셔너리 주요 메서드
```python
print(menu.keys()) # key 목록 조회
print(menu.values()) # value 목록 조회
print(menu.items()) # items 조회 (키,값) 튜플 형태로 리턴
list(menu.keys())[1] # 딕셔너리는 인덱서 방식으로 값을 참조 불가 > list로 변환
print(sum(menu.values())) # 정수나 실수의 값을 더해줌
print("coffee" in menu) # 딕셔너리에 coffee가 있기때문에 True 출력
print(4000 in menu) # key만 검색 가능(value 검색X)
print(4000 in menu.values()) # menu딕셔너리에 4000이란 값이 있다는면 True 출력
print(menu.get("cake")) # default 값이 없는 경우 None
```
### 집합

set_a = {1, 2, 3}
set_b = {3, 4, 5}

print(set_a & set_b) # 교집합 / 출력) {3}
print(set_a | set_b) # 합집합 / 출력) {1, 2, 3, 4, 5}

### 종합 실습
```python
# 회원 데이터
user_database = [
    {"name": "손흥민", "level": "VIP", "active": True, "point": 15000},
    {"name": "김민재", "level": "Gold", "active": False, "point": 9000},
    {"name": "조규성", "level": "VIP", "active": True, "point": 12000},
    {"name": "이강인", "level": "Gold", "active": True, "point": 8000},
    {"name": "홍명보", "level": "Bronze", "active": False, "point": 1000}
]

# VIP 포인트 누적 변수
total_vip_points = 0
# 현재 활성 회원 수
active_user_count = 0

print("=== 마케팅 대상 활성 VIP및 Gold 회원 결과")

for user in user_database:
    # 활성화 된 고객 필터링 active
    if user["active"]:
        # 활성 회원수 누적
        active_user_count += 1

        if user["level"] == "VIP":
            total_vip_points += user["point"]
            print(f"[VIP 등급] {user['name']} 회원님에게 VIP 쿠폰이 발행되었습니다.")
        elif user["level"] == "Gold":
            print(f"[Gold 등급] {user['name']} 회원님에게 Gold 쿠폰이 발행되었습니다.")

print(f"전체 활성 회원 수 : {active_user_count}명")
print(f"활성 VIP 포인트 누적 : {total_vip_points}점")
```
### 숫자 맞추기 게임
```python
import random # 난수 발생을 위한 모듈 임포트

# 1 ~ 100 사이의 난수 발생
number = random.randint(1, 10)

# 시도 횟수
try_count = 0

# 타이틀
print("="*42)
print("컴퓨터가 1부터 10사이의 숫자를 생성했습니다.")
print("="*42)

# 메인 루프
while True:
    user_input = input("컴퓨터의 숫자를 맞춰보세요. (1 ~ 10):")

    # 중도탈출 기능
    if user_input == "q":
        print(f"게임 종료합니다. 정답은 {number}입니다.")
        break

    if not user_input.isdecimal():
        print("숫자만 입력해주세요.")
        continue

    # 숫자 변환
    user_number = int(user_input)

    # 입력 숫자 범위 벗어났을 때
    if not (1 <= user_number <= 10):
        print(f"1 이상, 10 미만인 값만 입력해주세요.")
        continue
    
    # 시도 횟수 증가
    try_count += 1 # try_count++ , ++try_count ← 이런 문법들은 python에서 지원하지 않음

    # 입력값과 정답을 비교
    if user_number < number:
        print("Up! 더 큰 수 입니다.")
    elif user_number > number:
        print("Down! 더 작은 수 입니다.")
    else:
        print(f"정답입니다. {try_count}회 만에 성공했습니다.")
        break # 정답일 경우 무한 루프
```
